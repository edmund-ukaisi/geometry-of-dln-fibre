+import sympy as sp
+
+
+def entry_name(prefix, i, j):
+    return sp.Symbol(f"{prefix}{i}{j}")
+
+
+def adjudicate(n, m, a, b):
+    """Exact check for the zero-based pivot p=(a,b)."""
+    u = sp.Symbol("u")
+
+    # Generic post-blow-up unit matrix W: W[a,b] = 1.
+    W = sp.Matrix(n, m, lambda i, j:
+                  sp.Integer(1) if (i, j) == (a, b)
+                  else entry_name("w", i, j))
+
+    # Row and column elementary shears.  L clears pivot column; R clears
+    # pivot row.  Their non-identity parts square to zero.
+    L = sp.eye(n)
+    for i in range(n):
+        if i != a:
+            L[i, a] = -W[i, b]
+    R = sp.eye(m)
+    for j in range(m):
+        if j != b:
+            R[b, j] = -W[a, j]
+
+    normalized = sp.simplify(L * W * R)
+    residual = {
+        (i, j): sp.expand(W[i, j] - W[i, b] * W[a, j])
+        for i in range(n) if i != a
+        for j in range(m) if j != b
+    }
+    expected = sp.zeros(n, m)
+    expected[a, b] = 1
+    for (i, j), value in residual.items():
+        expected[i, j] = value
+
+    # N is the invertible chart-coordinate shear from normalized coordinates
+    # (u, alpha_i, beta_j, s_ij), placed in row-major matrix slots, to the
+    # ordinary chart coordinates (u, w_ij).  Off the pivot row/column,
+    # w_ij = s_ij + alpha_i*beta_j.
+    z_slots = []
+    chart_after_N = []
+    d_after_blowup_after_N = []
+    for i in range(n):
+        for j in range(m):
+            if (i, j) == (a, b):
+                z = u
+                chart = u
+                d = u
+            elif i == a:
+                z = entry_name("beta", a, j)
+                chart = z
+                d = u * chart
+            elif j == b:
+                z = entry_name("alpha", i, b)
+                chart = z
+                d = u * chart
+            else:
+                z = entry_name("s", i, j)
+                alpha = entry_name("alpha", i, b)
+                beta = entry_name("beta", a, j)
+                chart = z + alpha * beta
+                d = u * chart
+            z_slots.append(z)
+            chart_after_N.append(chart)
+            d_after_blowup_after_N.append(d)
+
+    J_N = sp.Matrix(chart_after_N).jacobian(z_slots)
+    J_composite = sp.Matrix(d_after_blowup_after_N).jacobian(z_slots)
+    det_N = sp.factor(J_N.det())
+    det_composite = sp.factor(J_composite.det())
+    expected_det = u ** (n * m - 1)
+
+    return {
+        "residual": residual,
+        "matrix_ok": normalized == expected,
+        "det_L": sp.factor(L.det()),
+        "det_R": sp.factor(R.det()),
+        "L_nil2": sp.simplify((L - sp.eye(n)) ** 2) == sp.zeros(n),
+        "R_nil2": sp.simplify((R - sp.eye(m)) ** 2) == sp.zeros(m),
+        "det_N": det_N,
+        "det_composite": det_composite,
+        "jacobian_ok": sp.simplify(det_composite - expected_det) == 0,
+    }
+
+
+all_ok = True
+for n, m in [(2, 2), (3, 3), (2, 3)]:
+    print(f"SIZE {n}x{m}")
+    for a in range(n):
+        for b in range(m):
+            ans = adjudicate(n, m, a, b)
+            all_ok &= (ans["matrix_ok"] and ans["det_L"] == 1
+                       and ans["det_R"] == 1 and ans["L_nil2"]
+                       and ans["R_nil2"] and ans["det_N"] == 1
+                       and ans["jacobian_ok"])
+            print(
+                f" p=({a},{b}) interior={a > 0 and b > 0}: "
+                f"residual={ans['residual']}; matrix_ok={ans['matrix_ok']}; "
+                f"detL={ans['det_L']}, detR={ans['det_R']}, "
+                f"nil2=({ans['L_nil2']},{ans['R_nil2']}); "
+                f"detN={ans['det_N']}; detComposite={ans['det_composite']}; "
+                f"jacobian_ok={ans['jacobian_ok']}"
+            )
+print(f"ALL_19_PIVOTS_OK={all_ok}")
+

Use zero-based indices and write \(D=uW\), where \(W_{ab}=1\). Define

\[
L_p=I_n-\sum_{i\ne a}w_{ib}E_{ia},
\qquad
R_p=I_m-\sum_{j\ne b}w_{aj}E_{bj}.
\]

Both are unipotent:

\[
(L_p-I)^2=(R_p-I)^2=0,\qquad \det L_p=\det R_p=1.
\]

Then

\[
L_pW R_p=
\begin{cases}
1 &(i,j)=(a,b),\\
0 &i=a,\ j\ne b,\\
0 &i\ne a,\ j=b,\\
s_{ij}=w_{ij}-w_{ib}w_{aj} &i\ne a,\ j\ne b.
\end{cases}
\]

Thus

\[
L_pDR_p=u
\begin{pmatrix}
\text{isolated pivot }1 & 0\\
0 & (s_{ij})
\end{pmatrix}.
\]

The regular chart-coordinate normalization is the polynomial shear

\[
w_{ib}=\alpha_i,\quad
w_{aj}=\beta_j,\quad
w_{ij}=s_{ij}+\alpha_i\beta_j.
\]

Its inverse is \(s_{ij}=w_{ij}-w_{ib}w_{aj}\), so it is a genuine polynomial coordinate change.
