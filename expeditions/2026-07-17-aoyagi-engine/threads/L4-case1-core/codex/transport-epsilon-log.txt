+import sympy as sp

+# Fresh symbols and matrices.
+u = {(L, r, c): sp.Symbol(f"u_{L}{r}{c}")
+     for L in range(3) for r in range(2) for c in range(2)}
+a, b, c, d = (u[0, 0, 0], u[0, 0, 1], u[0, 1, 0], u[0, 1, 1])
+x, y, z, w = (u[1, 0, 0], u[1, 0, 1], u[1, 1, 0], u[1, 1, 1])
+P, R, T, V = (u[2, 0, 0], u[2, 0, 1], u[2, 1, 0], u[2, 1, 1])
+A1 = sp.Matrix([[x, y], [z, w]])
+A2 = sp.Matrix([[P, R], [T, V]])
+
+def shear(vals, layer, cleared):
+    out = dict(vals)
+    for (L, r, col) in vals:
+        if L == layer and cleared < r and cleared < col:
+            out[L, r, col] = sp.expand(
+                vals[L, r, col] - vals[L, r, cleared] * vals[L, cleared, col])
+    return out
+
+def quotient(vals, pivot):
+    return {k: (sp.Integer(1) if k == pivot else v) for k, v in vals.items()}
+
+def blowup(vals, center, pivot):
+    if pivot is None:
+        return dict(vals)
+    return {k: (vals[pivot] if k == pivot else
+                vals[pivot] * v if k in center else v)
+            for k, v in vals.items()}
+
+def ed1(vals):
+    return quotient(shear(vals, 0, 0), (0, 0, 0))
+
+def ed2(vals):
+    return blowup(shear(vals, 0, 1), {(0, 1, 1)}, (0, 1, 1))
+
+def ed3(vals):
+    return blowup(vals, set(), None)
+
+def changes(edge):
+    image = edge(u)
+    return {str(u[k]): sp.expand(image[k]) for k in u if sp.expand(image[k] - u[k]) != 0}
+
+def entries(mat):
+    return [sp.expand(mat[i, j]) for i in range(2) for j in range(2)]
+
+def term_list(expr):
+    return sp.Add.make_args(sp.expand(expr)) if expr != 0 else ()
+
+def cdegree(term, center):
+    powers = term.as_powers_dict()
+    return sum(int(powers.get(v, 0)) for v in center)
+
+def tests(mat, center, extras, pivot):
+    result = []
+    zero_center = {v: 0 for v in center}
+    for f in entries(mat):
+        a1_break = sp.expand(f.subs(zero_center))
+        a2_terms = [q for q in term_list(f) if cdegree(q, center) > 1]
+        a3_break = {str(e): sp.expand(f.coeff(e).subs(pivot, 0))
+                    for e in extras if sp.expand(f.coeff(e).subs(pivot, 0)) != 0}
+        result.append({
+            "A1": a1_break == 0,
+            "A1_break": a1_break,
+            "A2": not a2_terms,
+            "A2_break": a2_terms,
+            "A3": not a3_break,
+            "A3_break": a3_break,
+        })
+    return result
+
+def print_tests(label, mat, center, extras, pivot):
+    got = tests(mat, center, extras, pivot)
+    print(label)
+    for i, row in enumerate(got):
+        print(f"  m{i}: {row}")
+    print("  OVERALL =", all(r["A1"] and r["A2"] and r["A3"] for r in got))
+
+# Literal recursion: ed1(ed2(ed3(u))).
+mapped = ed1(ed2(ed3(u)))
+A0_literal = sp.Matrix([[mapped[0, 0, 0], mapped[0, 0, 1]],
+                        [mapped[0, 1, 0], mapped[0, 1, 1]]])
+M_literal = A2 * A1 * A0_literal
+print("1 LITERAL MAPS")
+print("ed1 changes =", changes(ed1))
+print("ed2 changes =", changes(ed2))
+print("ed3 changes =", changes(ed3))
+print("A0 image =", A0_literal)
+print("foldResid =", entries(M_literal))
+print_tests("2 LITERAL TESTS; C=(u_011,u_100,u_110), extra=(u_101,u_111)",
+            M_literal, [d, x, z], [y, w], d)
+
+# Ingredient checks.  s is the Schur coordinate of normalized [[1,b],[c,d]].
+# h is the Schur coordinate of the *already literal-mapped* [[1,b],[c,d-b*c]].
+s, h = sp.symbols("s h")
+M_schur_only = sp.expand(M_literal.subs(d, s + b*c))
+print_tests("3a SCHUR PIVOT ONLY on literal residual (d=s+b*c)",
+            M_schur_only, [s, x, z], [y, w], s)
+
+B = sp.Matrix([[x, y], [z, w]])
+upper_literal_corner = sp.Matrix([[1, b], [0, d - 2*b*c]])
+M_column_literal_corner = A2 * B * upper_literal_corner
+print_tests("3b COLUMN CLEAR/DEEPER COORDS, but old literal pivot d",
+            M_column_literal_corner, [d, x, z], [y, w], d)
+
+upper_h = sp.Matrix([[1, b], [0, h]])
+diag_h = sp.diag(1, h)
+M_upper_h = A2 * B * upper_h
+M_diag_h = A2 * B * diag_h
+print("3c corrected-from-given foldResid (h=d-2*b*c) =", entries(M_upper_h))
+print_tests("3c COLUMN CLEAR + DEEPER COORDS + actual Schur pivot h",
+            M_upper_h, [h, x, z], [y, w], h)
+print_tests("3d FULL ROW+COLUMN CLEAR + actual Schur pivot h",
+            M_diag_h, [h, x, z], [y, w], h)
+
+# Faithful Gaussian clear starting from the normalized original A0.
+A0_original = sp.Matrix([[1, b], [c, d]])
+Q = sp.Matrix([[1, 0], [-c, 1]])
+Qinv = Q.inv()
+U = sp.Matrix([[1, -b], [0, 1]])
+Uinv = U.inv()
+S = sp.simplify(Q * A0_original)
+D = sp.simplify(S * U)
+S_s = S.subs(d, s + b*c)
+D_s = D.subs(d, s + b*c)
+A1_old_in_B = B * Q
+M_faithful = A2 * B * S_s
+M_full_clear = A2 * B * D_s
+print("4 FAITHFUL FACTORS")
+print("Q*A0 =", S)
+print("Q*A0*U =", D)
+print("B = A1_old*Qinv check =", sp.simplify(A1_old_in_B * Qinv - B) == sp.zeros(2))
+print("old product = B*(Q*A0) check =",
+      sp.simplify(A2*A1_old_in_B*A0_original.subs(d, s+b*c) - M_faithful) == sp.zeros(2))
+print("row-clear/output-factor check B*S = B*D*Uinv =",
+      sp.simplify(M_faithful - A2*B*D_s*Uinv) == sp.zeros(2))
+print("faithful residual =", entries(M_faithful))
+print_tests("4a FAITHFUL UPPER-TRIANGULAR RESIDUAL; C=(s,x,z)",
+            M_faithful, [s, x, z], [y, w], s)
+print_tests("4b FULLY CLEARED OUTPUT COORDS; C=(s,x,z)",
+            M_full_clear, [s, x, z], [y, w], s)
